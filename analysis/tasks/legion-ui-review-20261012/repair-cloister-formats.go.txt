package main
import("crypto/sha256";"dfolan/internal/catalog/pvf";"encoding/binary";"encoding/hex";"encoding/json";"fmt";"io";"os";"path/filepath";"runtime";"strings";"unicode/utf16")
type Row struct{Reference string `json:"reference"`;Original string `json:"original"`;Replacement string `json:"replacement"`}
type Patch struct{Source string `json:"source_sha256"`;Exe string `json:"exe_sha256"`;Rows []Row `json:"rows"`}
func must(e error){if e!=nil{panic(e)}}
func hash(p string)string{f,e:=os.Open(p);must(e);defer f.Close();h:=sha256.New();_,e=io.Copy(h,f);must(e);return hex.EncodeToString(h.Sum(nil))}

for _,row:=range patch.Rows{ref,ok:=pvf.ParseLocalizedRef(row.Reference);if !ok||ref.Table!=20{panic("unexpected table")};found:=0;for i,line:=range lines{cr:="";if line[at+1:]!=row.Original{panic("source text changed: "+ref.Key)};lines[i]=line[:at+1]+row.Replacement+cr;found++};if found!=1{panic("key count changed: "+ref.Key)};changed+=found};if changed!=5{panic("repair scope changed")};newText:=strings.Join(lines,"\n");newUnits:=utf16.Encode([]rune(newText));replacement:=make([]byte,len(newUnits)*2);for i,v:=range newUnits{binary.LittleEndian.PutUint16(replacement[i*2:],v)};inner:=filepath.Join(out,"Script.inner.pvf");fmt.Printf("TEXT_ENTRY path=%s old_bytes=%d new_bytes=%d format=%s\n",table,len(raw),len(replacement),a.Format());must(a.WriteEntryCopy(inner,table,replacement));count:=a.FileCount();a.Close();a=nil;runtime.GC();verify,e:=pvf.OpenReadOnly(pvf.Options{Path:inner,MaxBytes:2<<30},"");must(e);if !found||actual!=row.Replacement{panic("replacement verification failed")}};check,e:=verify.ReadRaw(table);must(e);if string(check)!=string(replacement){panic("table byte mismatch")};verify.Close();verify=nil;runtime.GC();outer:=filepath.Join(out,"Script.pvf");wrapped,e:=pvf.WrapOuter(client,inner,outer);must(e);h:=sha256.New();_,e=pvf.UnwrapOuterTo(client,outer,h);must(e);innerHash:=hash(inner);if hex.EncodeToString(h.Sum(nil))!=innerHash{panic("outer re-unwrapping differs")};record:=map[string]any{"source_inner_sha256":patch.Source,"table":table,"keys_changed":changed,"file_count_unchanged":count,"inner_sha256":innerHash,"outer_sha256":hash(outer),"outer_wrap":wrapped,"outer_unwrap_matches_inner":true,"installed":false};data,e:=json.MarshalIndent(record,"","  ");must(e);must(os.WriteFile(filepath.Join(out,"repair.json"),data,0600));fmt.Println(string(data))}
